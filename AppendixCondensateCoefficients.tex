\section{Condensate coefficient functions}
\label{app:condensate-coefficients}

For completeness, we give the coefficient functions used for the dimension-four
and dimension-six condensate contributions to the Borel moments.  We denote
\(\mathcal M\equiv M^2\) and
\begin{equation}
  \bar s(\xi)=\frac{m_b^2}{\xi}+\frac{m_c^2}{1-\xi}.
\end{equation}
The continuum-subtracted integration limits are
\begin{equation}
  \xi_\pm(s_0)=
  \frac{s_0+m_b^2-m_c^2\pm
  \sqrt{\lambda(s_0,m_b^2,m_c^2)}}{2s_0},
  \qquad
  \lambda(a,b,c)=a^2+b^2+c^2-2ab-2ac-2bc .
\end{equation}
The condensate pieces are evaluated as direct Borel transforms,
\begin{equation}
  \Pi^{ij}_{G^2}(\mathcal M,s_0)
  =
  \langle g_s^2G^2\rangle
  \int_{\xi_-}^{\xi_+}d\xi\,
  e^{-\bar s(\xi)/\mathcal M}
  C^{ij}_{G^2}(\xi,\mathcal M),
\end{equation}
\begin{equation}
  \Pi^{ij}_{G^3}(\mathcal M,s_0)
  =
  \langle g_s^3G^3\rangle
  \int_{\xi_-}^{\xi_+}d\xi\,
  e^{-\bar s(\xi)/\mathcal M}
  C^{ij}_{G^3}(\xi,\mathcal M),
\end{equation}
where \(ij=AA,AB,BB\).  The \(G^3\) coefficient functions below correspond to
the standard vacuum-averaged single-line heavy-quark propagator contribution.
Open-field cross-line dimension-six terms are not included in these
coefficients.

\subsection{Dimension-four gluon condensate}

It is convenient to write the coefficient functions as
\begin{align}
  C^{AA}_{G^2} &=
  \frac{P^{AA}_{G^2}}{288\pi^2\mathcal M^2(\xi-1)^3\xi^3},
  &
  C^{AB}_{G^2} &=
  \frac{P^{AB}_{G^2}}
  {96\pi^2\mathcal M^2(\xi-1)^3\xi^3(m_b+m_c)}, \nonumber\\
  C^{BB}_{G^2} &=
  \frac{P^{BB}_{G^2}}
  {288\pi^2\mathcal M^2(\xi-1)^4\xi^4(m_b+m_c)^2}.
\end{align}
The corresponding numerators are
\begin{align}
P^{AA}_{G^2}={}&
\xi^2\!\left[-\mathcal M^2(\xi-1)^3(5\xi+2)
+\mathcal M m_c^2(8\xi^3-15\xi^2+9\xi-2)
+3m_c^4\xi^2\right] \nonumber\\
&-m_b^2\xi(\xi-1)
\left[\mathcal M(\xi-1)^2(8\xi-5)
+3m_c^2(2\xi^2-2\xi+1)\right] \nonumber\\
&+3m_bm_c\xi(\xi-1)
\left[\mathcal M(-4\xi^2+4\xi-2)+m_c^2\xi\right]
+3m_b^4(\xi-1)^4
-3m_b^3m_c\xi(\xi-1)^2,
\end{align}
\begin{align}
P^{AB}_{G^2}={}&
\mathcal M^2(\xi-1)\xi
\left[m_b(\xi-1)^2(\xi^2-4\xi+2)
-m_c\xi^2(\xi^2+2\xi-1)\right] \nonumber\\
&+\mathcal M\Big[
m_b^3(\xi^2+2\xi-2)(\xi-1)^3
-m_b^2m_c\xi(\xi^4-7\xi^3+11\xi^2-7\xi+2) \nonumber\\
&\hspace{2.0cm}
-m_bm_c^2\xi(\xi^4+2\xi^3-7\xi^2+6\xi-2)
+m_c^3\xi^3(\xi^2-4\xi+1)\Big] \nonumber\\
&-(m_b-m_c)\Big[
m_b^4(\xi-1)^4
-m_b^3m_c\xi(\xi-1)^2
+m_b^2m_c^2\xi(-2\xi^3+4\xi^2-3\xi+1) \nonumber\\
&\hspace{2.0cm}
+m_bm_c^3\xi^2(\xi-1)
+m_c^4\xi^4\Big],
\end{align}
\begin{align}
P^{BB}_{G^2}={}&
m_b^2(\xi-1)\xi^2
\left[2\mathcal M^2(\xi-1)^3(3\xi+1)
+2\mathcal M m_c^2(11\xi^3-18\xi^2+9\xi-2) \right. \nonumber\\
&\hspace{0.8cm}\left.
+3m_c^4(3\xi^2-2\xi+1)\right] \nonumber\\
&+3m_bm_c(\xi-1)\xi^2
\left[2\mathcal M^2(3\xi^3-6\xi^2+4\xi-1)
+2\mathcal M m_c^2(2\xi^2-2\xi+1)-m_c^4\xi\right] \nonumber\\
&+m_c^2\xi^3
\left[-2\mathcal M^2(\xi-1)^2(3\xi^2+4\xi-4)
+\mathcal M m_c^2(-11\xi^3+6\xi^2+9\xi-4)
-3m_c^4\xi^2\right] \nonumber\\
&-m_b^4(\xi-1)^2\xi
\left[\mathcal M(\xi-1)^2(11\xi-8)
+3m_c^2(3\xi^2-4\xi+2)\right] \nonumber\\
&+6m_b^3m_c(\xi-1)^2\xi
\left[\mathcal M(-2\xi^2+2\xi-1)+m_c^2\xi\right]
+3m_b^6(\xi-1)^5
-3m_b^5m_c(\xi-1)^3\xi .
\end{align}
